# 移动端 Web 适配（mobile2）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在 feat/mobile2 分支上为现有 myPDM Web 应用实现纯只读的移动端视图层（<768px），桌面端零改动、同路由切换。

**Architecture:** 桌面代码完全不修改；新增 `frontend/src/mobile/` 独立移动视图层（移动外壳 + 每模块只读页面），复用现有 `services/api.ts`、`stores/`、`types/`、`can()` 权限。`App.tsx` 是唯一接线点：`useIsMobile()` 为 true 时渲染 `<MobileRoot>`（移动路由树，路径与桌面一致），否则渲染现有桌面 `<Layout>`。全面屏返回手势 = `history.back()`，应用以 popstate 为唯一返回信号，抽屉用 pushState 哨兵拦截。

**Tech Stack:** React 18 + TypeScript + Vite + Tailwind CSS + react-router-dom 6 + zustand + axios（全部现有，不新增依赖）。

**Spec:** `docs/superpowers/specs/2026-08-21-mobile2-design.md`

## Global Constraints

- 断点：`< 768px` 移动端，`≥ 768px` 桌面（`useIsMobile` 用 `'(max-width: 767px)'`）。
- **只读**：移动视图层只调用 `GET` 类 API，不出现任何 POST/PUT/DELETE 交互、表单、状态流转按钮。
- 桌面零改动：**不允许**修改 `frontend/src/components/Layout.tsx`、`frontend/src/pages/*` 现有桌面文件、`frontend/src/services/*`、`frontend/src/stores/*`。
- 唯一允许修改的现有文件：`frontend/src/App.tsx`（仅加移动分支，约 10 行）。
- 不新增 npm 依赖（含 devDependencies）；测试仅用现有 vitest node 环境（`src/**/*.test.ts`），纯逻辑抽函数测试，组件以 `npm run build` + 真机/DevTools 验证。
- 组件代码用 Tailwind 原子类；可点区域 ≥ 40px；长文本 `break-all` 或省略号。
- 每个任务结束：`npm run build` 通过（`cd frontend` 下执行），纯逻辑任务 `npm run test` 通过，然后 git commit。

```ts
import { useEffect, useState } from 'react';

export const MOBILE_QUERY = '(max-width: 767px)';

export function subscribeMediaQuery(query: string, onChange: (matches: boolean) => void): () => void {
  if (typeof window === 'undefined' || typeof window.matchMedia !== 'function') {
    return () => {};
  }
  const mql = window.matchMedia(query);
  onChange(mql.matches);
  const handler = (e: MediaQueryListEvent) => onChange(e.matches);
  mql.addEventListener('change', handler);
  return () => mql.removeEventListener('change', handler);
}

export function useIsMobile(): boolean {
  const [isMobile, setIsMobile] = useState<boolean>(
    () => typeof window !== 'undefined' && typeof window.matchMedia === 'function'
      ? window.matchMedia(MOBILE_QUERY).matches
      : false,
  );
  useEffect(() => subscribeMediaQuery(MOBILE_QUERY, setIsMobile), []);
  return isMobile;
}
```

  - `backInterceptReducer(state: BackLayer, action: { type: 'open-drawer'; drawerId: string } | { type: 'close-drawer' } | { type: 'pop' }): BackLayer`
    - `open-drawer` → `{ kind: 'drawer', drawerId }`
    - `close-drawer` → `{ kind: 'page' }`
    - `pop`：当前是 drawer 时关闭抽屉（`{ kind: 'page' }`）；当前是 page 时返回原状态（页面后退由路由处理，不在此状态机内）

`frontend/src/mobile/hooks/backIntercept.ts`：

```ts
export type BackLayer = { kind: 'page' } | { kind: 'drawer'; drawerId: string };

export type BackAction =
  | { type: 'open-drawer'; drawerId: string }
  | { type: 'close-drawer' }
  | { type: 'pop' };

export function backInterceptReducer(state: BackLayer, action: BackAction): BackLayer {
  switch (action.type) {
    case 'open-drawer':
      return { kind: 'drawer', drawerId: action.drawerId };
    case 'close-drawer':
      return { kind: 'page' };
    case 'pop':
      return state.kind === 'drawer' ? { kind: 'page' } : state;
  }
}
```

`frontend/src/mobile/MobileLogin.tsx`（核心逻辑照搬 `Login.tsx` 的 `finishLogin` / providers 加载，仅布局改为移动样式）：

```tsx
import { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { authApi } from '../services/api';
import { useAuthStore } from '../stores/auth';
import { getFeishuProviderParam, isFeishuClient } from '../lib/feishu';

export default function MobileLogin() {
  const navigate = useNavigate();
  const [username, setUsername] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [feishuProviders, setFeishuProviders] = useState<{ key: string; name: string }[]>([]);
  const [wechatProviders, setWechatProviders] = useState<{ key: string; name: string }[]>([]);

  const finishLogin = async (accessToken: string, refreshToken?: string | null) => {
    useAuthStore.getState().setUser(null, accessToken);
    const userResponse = await authApi.me();
    useAuthStore.getState().setUser(userResponse.data, accessToken);
    navigate(userResponse.data.must_change_password ? '/change-password' : '/');
  };

  useEffect(() => {
    authApi.feishuConfig().then((res) => {
      const providers = res.data.providers ?? [];
      setFeishuProviders(providers);
      if (!isFeishuClient() || providers.length === 0) return;
      const want = getFeishuProviderParam();
      const provider = providers.find((p) => (want ? p.key === want : p.key === 'feishu')) ?? providers[0];
      window.location.href = `/api/auth/feishu/authorize?provider=${provider.key}`;
    });
    authApi.wechatConfig().then((res) => setWechatProviders(res.data.providers ?? []));
  }, []);

  const submit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError('');
    try {
      const res = await authApi.login(username, password);
      await finishLogin(res.data.access_token, res.data.refresh_token);
    } catch {
      setError('用户名或密码错误');
    }
  };

  return (
    <div className="h-full flex flex-col items-center justify-center px-6 bg-gray-50">
      <h1 className="text-2xl font-bold mb-1">myPDM</h1>
      <p className="text-sm text-gray-500 mb-8">移动端 · 只读查询</p>
      <form onSubmit={submit} className="w-full max-w-sm flex flex-col gap-3">
        <input className="h-12 px-4 rounded-lg bg-white border border-gray-200 text-base" placeholder="用户名" value={username} onChange={(e) => setUsername(e.target.value)} />
        <input className="h-12 px-4 rounded-lg bg-white border border-gray-200 text-base" type="password" placeholder="密码" value={password} onChange={(e) => setPassword(e.target.value)} />
        {error && <p className="text-sm text-red-600">{error}</p>}
        <button className="h-12 rounded-lg bg-primary text-white text-base font-medium">登录</button>
      </form>
      {(feishuProviders.length > 0 || wechatProviders.length > 0) && (
        <div className="w-full max-w-sm mt-6 flex flex-col gap-2">
          {feishuProviders.map((p) => (
            <button key={p.key} className="h-11 rounded-lg bg-white border border-gray-200 text-sm" onClick={() => (window.location.href = `/api/auth/feishu/authorize?provider=${p.key}`)}>
              {p.name ?? '飞书登录'}
            </button>
          ))}
          {wechatProviders.map((p) => (
            <button key={p.key} className="h-11 rounded-lg bg-white border border-gray-200 text-sm" onClick={() => (window.location.href = `/api/auth/wechat/authorize`)}>
              微信扫码登录
            </button>
          ))}
        </div>
      )}
    </div>
  );
}
```

import { useEffect, useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { partsApi } from '../../services/api';

import MobileCardList from '../components/MobileCardList';
import StatusBadge from '../components/StatusBadge';
import EmptyState from '../components/EmptyState';
import { formatMeta } from '../components/formatMeta';
import type { PartListItem } from '../../types';

const STATUS_MAP: Record<string, { label: string; cls: string }> = {
  draft: { label: '草稿', cls: 'bg-blue-100 text-blue-800' },
  frozen: { label: '冻结', cls: 'bg-orange-100 text-orange-800' },
  released: { label: '发布', cls: 'bg-green-100 text-green-800' },
  obsolete: { label: '作废', cls: 'bg-red-100 text-red-800' },
};

        renderMain={(p) => `${p.code} ${p.name}`}

          <span className="flex flex-wrap items-center gap-2">

        )}
